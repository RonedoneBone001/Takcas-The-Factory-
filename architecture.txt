MAX_SIZE = 8 * 1024 * 1024  # 8 МБ

# WHITELIST: только эти ссылки будут скачаны (максимум 3)
WHITELIST = [
    {
        "url": "https://raw.githubusercontent.com/termux/termux-api/master/README.md",
        "filename": "termux_api.md"
    },
    # Добавь до 2 других ссылок (если нужно)
    # {
    #     "url": "https://example.com/some_doc.md",
    #     "filename": "some_doc.md"
    # },
]

def download_file(url, filepath):
    """
    Скачивает файл с проверкой размера.
    Возвращает True (успех) / False (ошибка).
    """
    try:
        # Открываем URL (с таймаутом 30 сек)
        req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req, timeout=30) as response:
            # Читаем данные (с лимитом размера)
            data = response.read(MAX_SIZE + 1)  # Читаем на 1 байт больше лимита
            
            # Проверка размера
            if len(data) > MAX_SIZE:
                print(f"[DOCS] Ошибка: Файл слишком большой, отклонен (>{MAX_SIZE} байт)")
                return False
            
            # Записываем в файл
            with open(filepath, "wb") as f:
                f.write(data)
            
            print(f"[DOCS] Скачано: {filepath} ({len(data)} байт)")
            return True
    
    except Exception as e:
        print(f"[DOCS] Ошибка при скачивании {url}: {e}")
        return False

def main():

    except Exception as e:
        print(f"[DOCS] Ошибка при создании папки: {e}")
        return

    if len(WHITELIST) > 3:

        return

    # 3. Обрабатываем каждую ссылку
    for item in WHITELIST:
        url = item["url"]
        filename = item["filename"]
        filepath = os.path.join(DOCS_DIR, filename)
        
        print(f"\n[DOCS] Обработка: {url}")
        
        # 3.1 Проверка на повторное скачивание

            print(f"[DOCS] Файл уже существует: {filepath} (пропускаем)")

import threading
import time
from collections import OrderedDict


class TTLCache:
    def __init__(self, capacity, default_ttl, cleanup_interval):
        self.capacity = capacity
        self.default_ttl = default_ttl
        self.cleanup_interval = cleanup_interval
        self.cache = OrderedDict()
        self.lock = threading.RLock()
        self.hits = 0
        self.misses = 0
        self.evictions = 0
        self.expirations = 0
        self.on_evict = None
        self._stop_event = threading.Event()
        self._cleaner_thread = None

    def get(self, key):
        victims = []
        with self.lock:
            item = self.cache.get(key)
            if item is None:
                self.misses += 1
                result = None
            else:
                value, expires_at = item
                if expires_at is not None and time.monotonic() >= expires_at:
                    self.cache.pop(key)
                    self.expirations += 1
                    self.misses += 1
                    victims.append((key, value, "ttl"))
                    result = None
                else:
                    self.hits += 1
                    self.cache.move_to_end(key)
                    result = value
        self._fire_callbacks(victims)
        return result

    def put(self, key, value, ttl=None):
        victims = []
        ttl = self.default_ttl if ttl is None else ttl
        expires_at = (time.monotonic() + ttl) if ttl is not None else None
        with self.lock:
            if key in self.cache:
                self.cache.pop(key)
            self.cache[key] = (value, expires_at)
            while len(self.cache) > self.capacity:
                old_key, (old_value, _) = self.cache.popitem(last=False)
                self.evictions += 1
                victims.append((old_key, old_value, "lru"))
        self._fire_callbacks(victims)

    def get_stats(self):
        with self.lock:
            total = self.hits + self.misses
            return {
                "hits": self.hits,
                "misses": self.misses,
                "evictions": self.evictions,
                "expirations": self.expirations,
                "size": len(self.cache),
                "hit_ratio": self.hits / total if total else 0.0,
            }

    def set_on_evict(self, callback):
        with self.lock:
            self.on_evict = callback

    def start_cleanup(self):
        if self._cleaner_thread is not None:
            return
        self._cleaner_thread = threading.Thread(
            target=self._cleanup_loop, daemon=True
        )
        self._cleaner_thread.start()

    def stop_cleanup(self):
        self._stop_event.set()
        if self._cleaner_thread is not None:
            self._cleaner_thread.join(timeout=2.0)
            self._cleaner_thread = None

    def _cleanup_loop(self):
        while not self._stop_event.wait(self.cleanup_interval):
            self._cleanup_once()

    def _cleanup_once(self):
        now = time.monotonic()
        expired = []
        with self.lock:
            for key, (value, expires_at) in list(self.cache.items()):
                if expires_at is not None and now >= expires_at:
                    self.cache.pop(key)
                    self.expirations += 1
                    expired.append((key, value, "ttl"))
        self._fire_callbacks(expired)

    def _fire_callbacks(self, victims):
        cb = self.on_evict
        if cb is None:
            return
        for key, value, reason in victims:
            try:
                cb(key, value, reason)
            except Exception:
                pass
