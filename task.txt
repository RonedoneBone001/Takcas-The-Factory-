import threading
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity: int, on_evict=None):
        self.cache = OrderedDict()
        self.capacity = capacity
        self.lock = threading.Lock()
        self.hits = 0
        self.misses = 0
        self.evictions = 0
        self.on_evict = on_evict

    def get(self, key):
        with self.lock:
            if key in self.cache:
                self.hits += 1
                self.cache.move_to_end(key)
                return self.cache[key]
            else:
                self.misses += 1
                return None

    def put(self, key, value):
        evicted = None
        with self.lock:
            if key in self.cache:
                self.cache.move_to_end(key)
                self.cache[key] = value
            else:
                if len(self.cache) >= self.capacity:
                    evicted_key, evicted_value = self.cache.popitem(last=False)
                    self.evictions += 1
                    evicted = (evicted_key, evicted_value)
                self.cache[key] = value

        # Вызываем callback ВНЕ лока
        if evicted and self.on_evict:
            self.on_evict(evicted[0], evicted[1])

def test_lru_cache():
    callback_under_lock = threading.local()
    callback_under_lock.is_locked = False

    def on_evict(key, value):
        # Пытаемся захватить лок. Если получается - значит, мы ВНЕ лока.
        if cache.lock.acquire(blocking=False):
            cache.lock.release()
            callback_under_lock.is_locked = False
        else:
            callback_under_lock.is_locked = True

    cache = LRUCache(capacity=3, on_evict=on_evict)
    threads = []

    def worker():
        for i in range(100):
            key = i % 5
            if i % 2 == 0:
                cache.put(key, f"value{i}")
            else:
                cache.get(key)

    for _ in range(4):
        t = threading.Thread(target=worker)
        threads.append(t)
        t.start()

    for t in threads:
        t.join()

    print("Hits:", cache.hits)
    print("Misses:", cache.misses)
    print("Evictions:", cache.evictions)
    print("Callback called under lock:", callback_under_lock.is_locked)
    assert not callback_under_lock.is_locked, "Callback was called under lock!"
    print("Test passed successfully!")

if __name__ == "__main__":
    test_lru_cache()
